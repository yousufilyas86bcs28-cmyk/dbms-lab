-- EX4 HACKERRANK - NESTED QUERIES (SUBQUERIES) ONLY
 
-- 1. DEPARTMENT TOP 3 SALARIES (Nested Subquery)
SELECT
    (SELECT name FROM Department WHERE id = e.departmentId) AS Department,
    e.name AS Employee,
    e.salary AS Salary
FROM (
    SELECT *,
           (
               SELECT COUNT(DISTINCT e2.salary)
               FROM Employee e2
               WHERE e2.departmentId = e1.departmentId
                 AND e2.salary >= e1.salary
           ) AS rnk
    FROM Employee e1
) e
WHERE e.rnk <= 3
ORDER BY e.departmentId, e.salary DESC;
 
 
-- 2. DEPARTMENT WITH HIGHEST SALARY (Nested Subquery)
SELECT
    (SELECT name FROM Department WHERE id = e.departmentId) AS Department,
    e.name AS Employee,
    e.salary AS Salary
FROM Employee e
WHERE e.salary = (
    SELECT MAX(salary)
    FROM Employee
    WHERE departmentId = e.departmentId
);
 
 
-- 3. JAPANESE CITY (DQL - SELECT)
SELECT *
FROM CITY
WHERE COUNTRYCODE = 'JPN';
 
 
-- 4. WEATHER OBSERVATION STATION 20 (Nested Subquery)
SELECT ROUND(PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY LAT_N), 4)
FROM STATION
WHERE LAT_N IS NOT NULL;
 
 
-- 5. BINARY TREE NODES (Nested Subquery)
SELECT N,
    CASE
        WHEN P IS NULL THEN 'Root'
        WHEN N IN (SELECT P FROM BST WHERE P IS NOT NULL) THEN 'Inner'
        ELSE 'Leaf'
    END
FROM BST
ORDER BY N;
 
 
-- 6. TYPES OF TRIANGLE (DQL - SELECT / CASE)
SELECT
    CASE
        WHEN A + B <= C OR A + C <= B OR B + C <= A THEN 'Not A Triangle'
        WHEN A = B AND B = C                         THEN 'Equilateral'
        WHEN A = B OR B = C OR A = C                 THEN 'Isosceles'
        ELSE 'Scalene'
    END AS triangle_type
FROM TRIANGLES;
 
 
-- 7. OCCUPATIONS (Nested Subquery with ROW_NUMBER)
SELECT
    MAX(CASE WHEN Occupation = 'Doctor'    THEN Name END) AS Doctor,
    MAX(CASE WHEN Occupation = 'Professor' THEN Name END) AS Professor,
    MAX(CASE WHEN Occupation = 'Singer'    THEN Name END) AS Singer,
    MAX(CASE WHEN Occupation = 'Actor'     THEN Name END) AS Actor
FROM (
    SELECT
        Name,
        Occupation,
        ROW_NUMBER() OVER (PARTITION BY Occupation ORDER BY Name) AS rn
    FROM OCCUPATIONS
) ranked
GROUP BY rn
ORDER BY rn;
 
 
-- 8. OLIVANDER'S INVENTORY (Nested Subquery)
SELECT
    id,
    (SELECT age  FROM Wands_Property WHERE code = w.code) AS age,
    coins_needed,
    power
FROM Wands w
WHERE (SELECT is_evil FROM Wands_Property WHERE code = w.code) = 0
  AND w.coins_needed = (
      SELECT MIN(w2.coins_needed)
      FROM Wands w2
      WHERE w2.power = w.power
        AND (SELECT is_evil FROM Wands_Property WHERE code = w2.code) = 0
        AND (SELECT age   FROM Wands_Property WHERE code = w2.code)
          = (SELECT age   FROM Wands_Property WHERE code = w.code)
  )
ORDER BY w.power DESC,
         (SELECT age FROM Wands_Property WHERE code = w.code) DESC;
